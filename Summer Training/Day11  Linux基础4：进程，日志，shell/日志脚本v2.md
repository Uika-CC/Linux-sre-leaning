\#!/bin/bash

LOG_FILE=$1

total=$(wc -l < $LOG_FILE)

status_ma_count=$(awk '{print $NF}' $1)

status_count=$(awk '{print $NF}' $1 | sort | uniq -c)

IP_count=$(awk '{print $1}' $1 | sort | uniq -c | sort -nr)

echo "=====日志分析报告====="

echo ""

echo "1.总访问量:$total"

echo "2.状态码统计:$status_ma_count"

echo "3.各个状态码的请求次数:$status_count"

echo "4.访问最多的IP:$IP_count"

echo ""

echo "======================"
