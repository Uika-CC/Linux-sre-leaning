import psutil

def cpu_check():

​    cpu = psutil.cpu_percent(interval=1)

​    return {

​        "name":"cpu",

​        "value":cpu,

​        "status":cpu > 80

​    }

def memory_check():

​    mem = psutil.virtual_memory()

​    return {

​        "name":"memory",

​        "value":mem.percent,

​        "status":mem.percent > 80

​    }
