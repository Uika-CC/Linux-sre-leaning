\#!/bin/bash

LOG_FILE="$HOME/monitor/system_check.log"

echo "==============================" >> $LOG_FILE

echo "System Check Time: $(date)" >> $LOG_FILE

echo "==============================" >> $LOG_FILE

echo "[System Uptime]" >> $LOG_FILE

uptime >> $LOG_FILE

echo "" >> $LOG_FILE

echo "[Memory Usage]" >> $LOG_FILE

free -h >> $LOG_FILE

echo "" >> $LOG_FILE

echo "[Disk Usage]" >> $LOG_FILE

df -h >> $LOG_FILE

echo "" >> $LOG_FILE

echo "[Network Port]" >> $LOG_FILE

ss -tnlp >> $LOG_FILE

echo "" >> $LOG_FILE

echo "Check Finished"
